\subsection{infiltration\_coefficient\_map.txt}
\index{Source term!Infiltration}
\subsubsection{Purpose}
This file is required when \textbf{model\_runoff} is set to \textbf{infiltration\_coefficient}. In this case, the runoff rate $R$ is computed from the precipitation rate $P$ as
\begin{equation}
R = (1 - I_c) P
\end{equation} where $I_c \in [0, 1]$ is the infiltration coefficient. The spatial distribution for $I_c$ must be provided.
%
\subsubsection{Format description}
File format for a uniform $I_c$.  
\texttt{\textbf{Bold}} : keyword, must appear as such in the file. \texttt{<Tab>}: tabulation character. \\
    \begin{tabular}{p{6cm} | p{9cm}}
    \hline
    Field(s) & Comment(s) \\
    \hline  \hline
    \texttt{\textbf{Unif} <Tab> \textbf{1}} & Flag is 1 because $I_c$ is uniform \\ 
    \hline
    \texttt{\textbf{==== Default Param}} & Section separator line, leave unchanged \\
    \hline
    \texttt{\textbf{Infiltration\_coeff}} & Comment line, leave unchanged \\  
    \hline
    \texttt{k} & The uniform value for $I_c$ (dimensionless, between 0 and 1) \\ 
    \hline
    \texttt{\textbf{==== Distrib}} & Section separator line, leave unchanged \\  
    \hline
    \end{tabular}
%
\vspace{10mm} \\
%
File format for a spatially distributed $I_c$.  
\texttt{\textbf{Bold}} : keyword, must appear as such in the file. \texttt{<Tab>}: tabulation character. \\
    \begin{tabular}{p{6cm} | p{9cm}}
    \hline
    Field(s) & Comment(s) \\
    \hline  \hline
    \texttt{\textbf{Unif} <Tab> \textbf{0}} & Flag is 0 because $I_c$ is spatially variable \\ 
    \hline
    \texttt{\textbf{==== Default Param}} & Section separator line, leave unchanged \\ 
    \hline
    \texttt{\textbf{Infiltration\_coeff}} & Comment line, leave unchanged \\ 
    \hline
    \texttt{k}     & This value will be ignored because \texttt{Flag} = 1 \\ 
    \hline
    \texttt{\textbf{==== Distrib}} & Section separator line, leave unchanged \\  
    \hline
    \texttt{k[i]}      & Values for $I_c$ (dimensionless, between 0 and 1), provided on a cell-by-cell basis. Index i is the index of the cell in the mesh.  \\ 
    \hline
    \end{tabular}

\paragraph*{Sample file}
For a spatially ditributed coefficient.

\begin{verbatim}
Unif	0
==== Default Param
infiltration_coefficient
0.2
==== Distrib
0.12
0.15
0.35
\end{verbatim}
