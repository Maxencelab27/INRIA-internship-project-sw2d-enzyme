\subsection{infiltration\_h\_proportional\_parameter\_map.txt}
\index{Source term!Infiltration}
\subsubsection{Purpose}
This file is required when \textbf{model\_infiltration} is set to \textbf{h\_proportional}. In this case, the infilgtrataion rate $I$ is comuted as
\begin{equation}
I = k h
\end{equation} where $h$ is the water depth. The spatial distribution for $h$ must be provided.
%
\subsubsection{Format description}
File format for a uniform $k$.  
\texttt{\textbf{Bold}} : keyword, must appear as such in the file. \texttt{<Tab>}: tabulation character. \\
    \begin{tabular}{p{6cm} | p{9cm}}
    \hline
    Field(s) & Comment(s) \\
    \hline  \hline
    \texttt{\textbf{Unif} <Tab> \textbf{1}} & Flag is 1 because $k$ is uniform \\ 
    \hline
    \texttt{\textbf{==== Default Param}} & Section separator line, leave unchanged \\
    \hline
    \texttt{\textbf{Infiltration\_coeff}} & Comment line, leave unchanged \\  
    \hline
    \texttt{k} & The uniform values for $k$ (in s$^{-1}$) \\ 
    \hline
    \texttt{\textbf{==== Distrib}} & Section separator line, leave unchanged \\  
    \hline
    \end{tabular}
%
\vspace{10mm} \\
%
File format for a spatially distributed $k$.  
\texttt{\textbf{Bold}} : keyword, must appear as such in the file. \texttt{<Tab>}: tabulation character. \\
    \begin{tabular}{p{6cm} | p{9cm}}
    \hline
    Field(s) & Comment(s) \\
    \hline  \hline
    \texttt{\textbf{Unif} <Tab> \textbf{0}} & Flag is 0 because $k$ is spatially variable \\ 
    \hline
    \texttt{\textbf{==== Default Param}} & Section separator line, leave unchanged \\ 
    \hline
    \texttt{\textbf{Infiltration\_coeff}} & Comment line, leave unchanged \\ 
    \hline
    \texttt{k}     & This value will be ignored because \texttt{Flag} = 1 \\ 
    \hline
    \texttt{\textbf{==== Distrib}} & Section separator line, leave unchanged \\  
    \hline
    \texttt{k[i]}      & Values for $k$ (in s$^{-1}$), provided on a cell-by-cell basis. Index i is the index of the cell in the mesh.  \\ 
    \hline
    \end{tabular}
